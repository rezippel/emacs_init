#+TITLE: Emacs Init File
#+STARTUP: overview
#+PROPERTY: header-args:emacs-lisp :tangle yes

The "literate" version of my init file.

Startup flows through three files in =~/.emacs.d/=:
1. =~/.emacs.d/early-init.el=: Runs before the GUI frame or package manager initializes. It
   temporarily raises ~gc-cons-threshold~ so garbage collection does not slow down startup, disables
   ~package-enable-at-startup~, and suppresses the graphical toolbar before the initial frame is
   drawn.
2. =~/.emacs.d/init.el=: Checks whether ~config.org~ is newer than ~config.el~. It only tangles ~config.org~
   when needed, and otherwise loads the compiled ~config.el~ directly for fast startup.
3. =~/.emacs.d/config.org~ (this file): Contains all configuration organized by topic. Only blocks
   marked ~:tangle yes~ (the default via ~#+PROPERTY: header-args:emacs-lisp :tangle yes~) are written
   to ~config.el~; all surrounding Org prose and ~:tangle no~ blocks remain documentation-only.

Lexical binding is enabled at the very top of ~config.el~ because packages like ~consult~ and our custom
closures rely on lexical scoping rather than dynamic binding.
#+BEGIN_SRC emacs-lisp :tangle yes
;; -*- lexical-binding: t; -*-
;; (toggle-debug-on-error)   ;; for debugging init file.
#+END_SRC

* Core Emacs setup
** Utilities
First, some small helper functions used across the configuration:
- ~add-load-path~ appends an expanded directory path to the end of Emacs's ~load-path~.
- ~capitalize-words~ converts snake_case strings into Title Case with spaces.

#+BEGIN_SRC emacs-lisp :tangle yes
  (defun add-load-path (new-path)
    (add-to-list 'load-path (expand-file-name new-path)
                 t))

  (defun capitalize-words (title)
    (replace-regexp-in-string "_" " " (capitalize title)))
#+END_SRC

** Initialize the use-package mechanism
Configure ~package.el~ and ~use-package~ (which is built into Emacs 29+):
- Adds MELPA alongside the default GNU and NonGNU ELPA repositories.
- Prefers newer ~.el~ source files over stale byte-compiled ~.elc~ files (~load-prefer-newer~).
- Advises ~package-install~ (~rz/refresh-on-error~) so that if installing a newly added package fails
  due to a stale local MELPA archive cache (for example, after pulling ~config.org~ on another
  machine), Emacs automatically runs ~package-refresh-contents~ once and retries.
- Sets ~use-package-always-ensure t~ so declared packages are installed automatically unless ~:ensure
  nil~ is specified (for built-in packages like ~org~, ~flyspell~, or ~reftex~).
#+BEGIN_SRC emacs-lisp :tangle yes
  (require 'package)
  (setq package-enable-at-startup nil)
  (setq load-prefer-newer t)

  ;; GNU and NonGNU ELPA are enabled by default in Emacs 28+.
  ;; Add MELPA for additional community packages:
  (add-to-list 'package-archives '("melpa" . "https://melpa.org/packages/") t)

  (package-initialize)

  ;; Automatically refresh package archives once if installing a new package fails
  ;; due to a stale local MELPA index (e.g., after pulling config.org on another machine).
  (defvar rz/package-contents-refreshed nil)
  (define-advice package-install (:around (orig-fn pkg &rest args) rz/refresh-on-error)
    (condition-case err
        (apply orig-fn pkg args)
      (error
       (if rz/package-contents-refreshed
           (signal (car err) (cdr err))
         (setq rz/package-contents-refreshed t)
         (package-refresh-contents)
         (apply orig-fn pkg args)))))

  ;; `use-package' is built into Emacs 29+; install only on older Emacs.
  (unless (package-installed-p 'use-package)
    (package-refresh-contents)
    (package-install 'use-package))

  (require 'use-package)
  (setq use-package-verbose t
        use-package-always-ensure t)
#+END_SRC

** OS Environment

We maintain a list of feature symbols (~macos~, ~linux~, ~bash~, ~google~) in the ~os-environment~ variable so
later sections (fonts, frame geometry, TeX paths, and internal libraries) can branch cleanly on the
current host environment.

#+BEGIN_SRC emacs-lisp :tangle yes
  (defvar os-environment ()
    "Used to conditionalize loading of features based on the OS")

  (when (eq system-type 'darwin)
    (add-to-list 'os-environment 'macos))

  (when (eq system-type 'gnu/linux)
    (add-to-list 'os-environment 'linux))

  (when (and shell-file-name (string-match-p "bash" shell-file-name))
    (add-to-list 'os-environment 'bash))
#+END_SRC

Store backup (~file~~) and auto-save (~#file#~) files inside ~/.emacs.d/backups/~ and
~/.emacs.d/auto-save/~ rather than ~/tmp~ or cluttering working directories. Using ~version-control t~
with ~kept-new-versions 6~ and ~delete-old-versions t~ keeps the 6 most recent numbered backups of each
file and prunes older ones automatically without needing a startup cleanup loop.
#+BEGIN_SRC emacs-lisp :tangle yes
  (let ((backup-dir (expand-file-name "backups/" user-emacs-directory))
        (autosave-dir (expand-file-name "auto-save/" user-emacs-directory)))
    (make-directory backup-dir t)
    (make-directory autosave-dir t)
    (unless backup-inhibited
      (setq backup-directory-alist `((".*" . ,backup-dir))
            auto-save-file-name-transforms `((".*" ,autosave-dir t))
            backup-by-copying t
            version-control t
            kept-new-versions 6
            kept-old-versions 2
            delete-old-versions t)))
#+END_SRC

Automatically re-tangle ~config.org~ to ~config.el~ whenever ~config.org~ is saved inside Emacs. This
pairs with ~init.el~ so normal startups load ~config.el~ directly without loading ~org-babel~ first.
#+BEGIN_SRC emacs-lisp :tangle yes
  (defun rz/tangle-config-on-save ()
    "Automatically tangle `~/.emacs.d/config.org' when saved."
    (when (and buffer-file-name
               (file-equal-p buffer-file-name
                             (expand-file-name "config.org" user-emacs-directory)))
      (let ((org-confirm-babel-evaluate nil))
        (org-babel-tangle))))

  (add-hook 'after-save-hook #'rz/tangle-config-on-save)
#+END_SRC

Configure the GUI frame appearance and scrolling behavior when running in a graphical display
(~display-graphic-p~):
- Enables the clock in the mode line (~display-time~).
- Enables pixel-resolution frame resizing (~frame-resize-pixelwise~) and smooth trackpad scrolling
  (~pixel-scroll-precision-mode~, built into Emacs 29+).
- Sets initial window dimensions (~set-frame-size~ takes width in columns followed by height in rows)
  and hides the graphical toolbar (~tool-bar-mode -1~).
#+BEGIN_SRC emacs-lisp :tangle yes
  (display-time)

  (when (display-graphic-p)
    (setq-default visible-bell nil)
    (setq frame-resize-pixelwise t)

    (when (fboundp 'pixel-scroll-precision-mode)
      (pixel-scroll-precision-mode 1))

    (cond ((memq 'macos os-environment)
           (set-frame-size (selected-frame) 120 70)
           ;; The following doesn't seem to do anything in Crostini
           (set-frame-position (selected-frame) 0 0))
          (t
           (set-frame-size (selected-frame) 100 55)))
    ;; This eliminates the fancy toolbar that takes up a lot of space.
    (tool-bar-mode -1))
#+END_SRC

Diagnostic helper (disabled via ~:tangle no~) to inspect all frame geometry parameters and core face
colors in a temporary buffer (~*Frame Parameters*~).

  ;; Figure out what the parameters mean 
  (defun show-frame-parameters (&optional frame)
    "Show the parameters of the specified FRAME, 
     or the current frame if nil: name, size, colors, fonts, and faces."
    (interactive)
    (let ((faces '(bold bold-italic default highlight italic
                        region secondary-selection underline))

          (names '(name left top height width window-id minibuffer font
                        auto-raise auto-lower vertical-scroll-bars
                        horizontal-scroll-bars icon-type
                        foreground-color background-color  mouse-color
                        cursor-color border-color cursor-type
                        border-width internal-border-width unsplittable
                        visibility menu-bar-lines parent-id))
          (params (frame-parameters nil)))
      (with-output-to-temp-buffer "*Frame Parameters*"
        (mapcar #'(lambda (name)
                    (princ (format "%-24s %s\n" name (cdr (assq name params)))))
                names)
        (mapcar #'(lambda (facename)
                    (princ (format
                            "face %-19s background %-14s  foreground %-14s\n"
                            facename (face-background facename nil)
                            (face-foreground facename nil))))
                faces))))

Enable standard session persistence and ergonomic modes:
- ~savehist-mode~: Persists minibuffer history across Emacs sessions so ~vertico~ and ~consult~ sort
  frequently used candidates to the top.
- ~recentf-mode~: Tracks recently opened files so they appear in ~consult-buffer~ (~C-x b~) even after
  restarting Emacs.
- ~save-place-mode~: Restores the cursor to its last position when reopening any file.
- ~delete-selection-mode~: Typing while a region is highlighted replaces the selection (matching
  Google Docs).
- ~global-auto-revert-mode~: Automatically reloads buffers when files change on disk (e.g., via ~git
  pull~ or sync scripts).
- ~which-key-mode~: Built into Emacs 30; pops up a cheatsheet of available key completions after
  pressing any prefix key (like ~C-c n~, ~C-c d~, or ~M-s~).
- Unlocks ~narrow-to-region~ (~C-x n n~), ~eval-expression~ (~M-:~), and ~downcase-region~ (~C-x C-l~) so Emacs
  doesn't prompt for confirmation.
#+BEGIN_SRC emacs-lisp :tangle yes
  (defun pop-to-old-buffer ()
    (interactive)
    (let ((new-buf (completing-read "Buffer Name: "
                                    (mapcar #'buffer-name (buffer-list))
                                    nil t)))
      (switch-to-buffer new-buf)))

  (keymap-global-set "C-x C-m" #'write-modified-files)

  ;; Persist minibuffer history, recent files, and last cursor position across sessions
  (savehist-mode 1)
  (recentf-mode 1)
  (save-place-mode 1)

  ;; Replace active selection when typing (like Google Docs) and auto-reload externally modified files
  (delete-selection-mode 1)
  (global-auto-revert-mode 1)

  ;; Built-in keybinding discovery popup (Emacs 30+)
  (when (fboundp 'which-key-mode)
    (which-key-mode 1))

  (put 'narrow-to-region 'disabled nil)
  (put 'eval-expression 'disabled nil)
  (put 'downcase-region 'disabled nil)

  (setq max-lisp-eval-depth 2000)
#+END_SRC

** golink-ify
Interactive helper ~M-x linkify~ that detects internal Google shortlinks at point (~go/...~, ~cl/...~,
~b/...~, ~g/...~) and converts them in place into formatted links for the current buffer's mode:
- In ~org-mode~ (and derived modes): converts ~go/foo~ to ~[[https://go.corp.google.com/foo][go/foo]]~.
- In ~markdown-mode~ (and derived modes): converts ~go/foo~ to ~[go/foo](https://go.corp.google.com/foo)~.
- With a prefix argument (~C-u M-x linkify~): replaces the shortlink with the bare expanded URL.

#+BEGIN_SRC emacs-lisp :tangle yes
  (require 'thingatpt)
  (require 'cl-lib)

  (defun alg/linkify-pattern-private (&optional arg regexp-pattern url-format)
    "Detect REGEXP-PATTERN at point and replace it with a Markdown or Org link.

  To build the URL, extracts the 1st matched group from REGEXP-PATTERN and
  substitutes it for %s in URL-FORMAT.  If ARG is non-nil, replaces with the
  bare URL.  Returns nil if the pattern is not detected."
    (when (thing-at-point-looking-at regexp-pattern)
      (let* ((beg (match-beginning 0))
             (end (match-end 0))
             (label (buffer-substring-no-properties beg end))
             (path (buffer-substring-no-properties
                    (match-beginning 1)
                    (match-end 1)))
             (url (format url-format path)))
        (cond
         (arg
          (message "Replacing shortlink with %s" url)
          (replace-region-contents beg end (lambda () url))
          t)
         ((derived-mode-p 'markdown-mode)
          (message "Replacing with a Markdown link")
          (replace-region-contents beg end (lambda () (format "[%s](%s)" label url)))
          t)
         ((derived-mode-p 'org-mode)
          (message "Replacing with an Org link")
          (replace-region-contents beg end (lambda () (format "[[%s][%s]]" url label)))
          t)
         (t
          (message "Unsupported mode for linkify: %s" major-mode)
          nil)))))

  (defun linkify (&optional arg)
    "Detect Google shortlinks (go/, cl/, b/, g/) and replace with Markdown or Org links.
  If called with prefix ARG, replaces with the bare URL."
    (interactive "P")
    (let ((patterns-formats
           '("go/\\([A-Za-z0-9&_+-]+\\)" "https://go.corp.google.com/%s"
             "cl/\\([0-9]+\\)"           "https://critique.corp.google.com/cl/%s"
             "b/\\([0-9]+\\)"            "https://b.corp.google.com/issues/%s"
             "g/\\([A-Za-z0-9._-]+\\)"   "https://groups.google.com/a/google.com/g/%s")))
      (cl-loop for (pattern format-string) on patterns-formats by #'cddr
               when (alg/linkify-pattern-private arg pattern format-string)
               return t)))
#+END_SRC

* Flyspell
Enable on-the-fly spell checking via ~flyspell~ backed by ~hunspell~ (when ~hunspell~ is installed on the

- Uses the ~en_US~ UTF-8 dictionary.
- Hooks ~flyspell-mode~ into all prose buffers (~text-mode~, which includes ~org-mode~ and ~markdown-mode~)

#+BEGIN_SRC emacs-lisp :tangle yes

* Consult and Friends
We use the modular **Vertico + Marginalia + Orderless + Consult + Corfu/Cape + Embark** completion
stack, which builds directly on Emacs's native ~completing-read~ and ~completion-at-point~ APIs rather
than replacing them.

**Vertico** renders minibuffer completion candidates in a clean vertical list.
#+BEGIN_SRC emacs-lisp :tangle yes
  (use-package vertico
    :ensure t
    :custom (vertico-cycle nil)
    :init (vertico-mode)
    )
#+END_SRC


**Marginalia** adds rich right-aligned annotations in the minibuffer (such as docstrings, keybindings
 on ~M-x~ commands, file permissions/sizes/timestamps on ~C-x C-f~, and buffer modes on ~C-x b~). Pressing
 ~M-A~ in the minibuffer cycles between heavy, light, and no annotations.
#+BEGIN_SRC emacs-lisp :tangle yes
  (use-package marginalia
    :after vertico
    :ensure t
    :custom
    (marginalia-annotators '(marginalia-annotators-heavy marginalia-annotators-light nil))
    ;; Either bind `marginalia-cycle' globally or only in the minibuffer
    :bind (:map minibuffer-local-map
  	  ("M-A" . marginalia-cycle))

    :init  (marginalia-mode))
#+END_SRC

**Orderless** lets you filter minibuffer candidates using space-separated search terms (substrings,
 regular expressions, or initials) in any order. File path completion retains ~partial-completion~ so
 ~~/N/p/p~ expands naturally to ~~/Notes/policy/policy_db.org~.
#+BEGIN_SRC emacs-lisp :tangle yes
  (use-package orderless
    :ensure t
    :custom
    (completion-styles '(orderless basic))
    (completion-category-overrides
     '((file (styles basic partial-completion)))))
#+END_SRC

**Consult** provides enhanced search and navigation commands with live buffer previews:
- **Buffer & File Switching**: ~C-x b~ (~consult-buffer~, combining open buffers, recent files, bookmarks,
  and ~org-roam~ buffers narrowed with ~r SPC~), ~C-x r b~ (~consult-bookmark~), ~C-x p b~
  (~consult-project-buffer~).

* Org Agenda

Keep ~org-agenda~ fast across hundreds of Org-roam files by dynamically restricting ~org-agenda-files~
to only those notes that actually contain active (incomplete) ~TODO~ items (adapted from [[https://www.d12frosted.io/posts/2021-01-16-task-management-with-roam-vol5][Boris
Buliga's dynamic Org-roam agenda pattern]]):
- ~vulpea-project-p~ parses the current buffer's headlines and returns non-nil if any headline has an
  active ~todo~ state.
- ~vulpea-project-update-tag~ runs on ~before-save-hook~ (intentionally omitted from ~find-file-hook~ so
  ~consult~ previews do not modify buffers) to add or remove the ~:project:~ tag in ~#+filetags:~
  depending on whether open tasks remain in the file.
- ~vulpea-project-files~ queries the ~org-roam~ SQLite database for all files tagged ~project~, and
  ~vulpea-agenda-files-update~ refreshes ~org-agenda-files~ right before ~org-agenda~ or ~org-todo-list~
  runs.
#+BEGIN_SRC emacs-lisp :tangle yes
;; From https://gist.github.com/d12frosted/a60e8ccb9aceba031af243dff0d19b2e


(defun vulpea-project-p ()
  "Return non-nil if current buffer has any todo entry.
TODO entries marked as done are ignored, meaning the this
function returns nil if current buffer contains only completed
tasks."
  (seq-find                                 ; (3)
   (lambda (type)
     (eq type 'todo))
   (org-element-map                         ; (2)
       (org-element-parse-buffer 'headline) ; (1)
       'headline
     (lambda (h)
       (org-element-property :todo-type h)))))

(defun vulpea-project-update-tag ()
    "Update PROJECT tag in the current buffer."
    (when (and (not (active-minibuffer-window))
               (vulpea-buffer-p))
      (save-excursion
        (goto-char (point-min))
        (let* ((tags (vulpea-buffer-tags-get))
               (original-tags tags))
          (if (vulpea-project-p)
              (setq tags (cons "project" tags))
            (setq tags (remove "project" tags)))

          ;; cleanup duplicates
          (setq tags (seq-uniq tags))

          ;; update tags if changed
          (when (or (seq-difference tags original-tags)
                    (seq-difference original-tags tags))
            (apply #'vulpea-buffer-tags-set tags))))))

(defun vulpea-buffer-p ()
  "Return non-nil if the currently visited buffer is a note."
  (and buffer-file-name
       (derived-mode-p 'org-mode)
       (string-suffix-p "org" buffer-file-name)
       (string-prefix-p
        (expand-file-name (file-name-as-directory org-roam-directory))
        (file-name-directory buffer-file-name))))

(defun vulpea-project-files ()
    "Return a list of note files containing 'project' tag." ;
    (seq-uniq
     (seq-map
      #'car
      (org-roam-db-query
       [:select [nodes:file]
        :from tags
        :left-join nodes
        :on (= tags:node-id nodes:id)
        :where (like tag (quote "%\"project\"%"))]))))

(defun vulpea-agenda-files-update (&rest _)
  "Update the value of `org-agenda-files'."
  (setq org-agenda-files (vulpea-project-files)))

;; Update `:project:` filetag on save (not on `find-file-hook', so `consult' live previews stay fast)
(add-hook 'before-save-hook #'vulpea-project-update-tag)

(advice-add 'org-agenda :before #'vulpea-agenda-files-update)
(advice-add 'org-todo-list :before #'vulpea-agenda-files-update)
(add-to-list 'org-tags-exclude-from-inheritance "project")
#+END_SRC

Supporting buffer-property functions borrowed from ~vulpea-buffer.el~ for reading, adding, updating,
and removing ~#+filetags:~ and other ~#+KEY: VALUE~ header keywords at the top of an Org buffer.
#+BEGIN_SRC emacs-lisp :tangle yes
;; functions borrowed from `vulpea' library
;; https://github.com/d12frosted/vulpea/blob/6a735c34f1f64e1f70da77989e9ce8da7864e5ff/vulpea-buffer.el

(defun vulpea-buffer-tags-get ()
  "Return filetags value in current buffer."
  (vulpea-buffer-prop-get-list "filetags" "[ :]"))

(defun vulpea-buffer-tags-set (&rest tags)
  "Set TAGS in current buffer.
If filetags value is already set, replace it."
  (if tags
      (vulpea-buffer-prop-set
       "filetags" (concat ":" (string-join tags ":") ":"))
    (vulpea-buffer-prop-remove "filetags")))

(defun vulpea-buffer-tags-add (tag)
  "Add a TAG to filetags in current buffer."
  (let* ((tags (vulpea-buffer-tags-get))
         (tags (append tags (list tag))))
    (apply #'vulpea-buffer-tags-set tags)))

(defun vulpea-buffer-tags-remove (tag)
  "Remove a TAG from filetags in current buffer."
  (let* ((tags (vulpea-buffer-tags-get))
         (tags (delete tag tags)))
    (apply #'vulpea-buffer-tags-set tags)))

(defun vulpea-buffer-prop-set (name value)
  "Set a file property called NAME to VALUE in buffer file.
If the property is already set, replace its value."
  (setq name (downcase name))
  (org-with-point-at 1
    (let ((case-fold-search t))
      (if (re-search-forward (concat "^#\\+" name ":\\(.*\\)")
                             (point-max) t)
          (replace-match (concat "#+" name ": " value) 'fixedcase)
        (while (and (not (eobp))
                    (looking-at "^[#:]"))
          (if (save-excursion (end-of-line) (eobp))
              (progn
                (end-of-line)
                (insert "\n"))
            (forward-line)
            (beginning-of-line)))
        (insert "#+" name ": " value "\n")))))

(defun vulpea-buffer-prop-set-list (name values &optional separators)
  "Set a file property called NAME to VALUES in current buffer.
VALUES are quoted and combined into single string using
`combine-and-quote-strings'.
If SEPARATORS is non-nil, it should be a regular expression
matching text that separates, but is not part of, the substrings.
If nil it defaults to `split-string-default-separators', normally
\"[ \f\t\n\r\v]+\", and OMIT-NULLS is forced to t.
If the property is already set, replace its value."
  (vulpea-buffer-prop-set
   name (combine-and-quote-strings values separators)))

(defun vulpea-buffer-prop-get (name)
  "Get a buffer property called NAME as a string."
  (org-with-point-at 1
    (when (re-search-forward (concat "^#\\+" name ": \\(.*\\)")
                             (point-max) t)
      (buffer-substring-no-properties
       (match-beginning 1)
       (match-end 1)))))

(defun vulpea-buffer-prop-get-list (name &optional separators)
  "Get a buffer property NAME as a list using SEPARATORS.
If SEPARATORS is non-nil, it should be a regular expression
matching text that separates, but is not part of, the substrings.
If nil it defaults to `split-string-default-separators', normally
\"[ \f\t\n\r\v]+\", and OMIT-NULLS is forced to t."
  (let ((value (vulpea-buffer-prop-get name)))
    (when (and value (not (string-empty-p value)))
      (split-string-and-unquote value separators))))

(defun vulpea-buffer-prop-remove (name)
  "Remove a buffer property called NAME."
  (org-with-point-at 1
    (when (re-search-forward (concat "\\(^#\\+" name ":.*\n?\\)")
                             (point-max) t)
      (replace-match ""))))
#+END_SRC
